boto3==1.40.72

# read_url — the URL reader. Both are ALWAYS lazy-imported inside
# runtime/url_reader.py (never at module load), exactly like boto3
# above, so a missing or broken wheel can NEVER brick boot; the tool degrades to
# a `status: error` and web_search still works.
#
# curl_cffi is THE fetcher (not a fallback for hard sites — measured across 7
# sites it returned more article text than aiohttp on every one, so aiohttp is
# now only the degraded path for a pod whose curl_cffi wheel is broken; the
# table is in url_reader.py). It is also the only reason this lane works
# without a browser: on a Cloudflare/Akamai-fronted site the 403 lands at
# the TLS HANDSHAKE, so a browser User-Agent on python's TLS stack changes
# nothing — libcurl-impersonate presents a real Chrome TLS + HTTP/2 fingerprint.
# It also reads http_proxy/https_proxy from the environment on its own
# (verified against a dead port), which is what keeps every fetch on
# the pod's governed egress squid.
#
# trafilatura does HTML→markdown with no browser. Its dependency tail (babel via
# courlan, dateparser via htmldate) is heavier than it looks — see below — and
# was paid deliberately: readability-lxml is ~40 MB cheaper but on the TLS
# wikipedia article it silently returned a 30k-char FRAGMENT from section 26
# where trafilatura returned all 267k from the lead. Silent truncation in text a
# model then reasons over is worse than the megabytes.
#
# MEASURED on the pinned python:3.14-slim-bookworm base: these two
# lines add 95 MB to the image (343 MB → 438 MB of site-packages) — curl_cffi
# 38 MB of impersonation binaries, babel 33 MB of locale data, lxml 12 MB, the
# rest small. Every transitive dep resolves to a cp314 wheel (installs clean
# under --only-binary=:all:), so unlike the Pillow incident above there is no
# source build and no compiler dependency.
curl_cffi==0.16.1
trafilatura==2.2.0

# Pictures for the messaging connectors (server/connectors/picture.py):
# pytakumi, the Python binding of Takumi (github.com/kane50613/takumi), draws a
# small HTML+CSS card with no browser: Rust, with its own HTML reader (html5ever),
# CSS engine (Servo's cssparser and selectors), layout (taffy), text (parley,
# harfrust) and rasteriser (tiny-skia), all compiled into one extension module
# with no system library beyond glibc and libgcc_s. Only the renderer's child process
# (server/connectors/_picture_child.py) imports it, never the kernel process,
# so a broken wheel costs pictures and nothing else.
#
# Pinned exactly: the renderer's CSS allowlist was measured against this
# version, and the in-image gate (scripts/test_connector_picture_image.py)
# measures it again on every build. pytakumi is an unofficial binding with a
# single maintainer (upstream Takumi publishes Node, WASM and Rust bindings
# only); 0.1.5 compiles in Takumi 2.14.0. It ships cp314 manylinux_2_28 wheels
# for x86_64 and aarch64 and needs no other package.
#
# It replaced html2pic + pictex + skia-python (and numpy, beautifulsoup4,
# tinycss2, stretchable, uharfbuzz, python-bidi under them) on 2026-10-19,
# because of the licence: the skia-python wheel has Adobe's DNG SDK compiled
# in, whose licence cannot ship with this AGPL-3.0-only tree. Takumi and its
# crates are permissive except four under MPL-2.0 (THIRD-PARTY-NOTICES.md).
#
# Takumi draws more of CSS than html2pic did, and differs from a browser in
# one default the renderer corrects (boxes are border-box; picture.py puts the
# page's tags back to content-box). It has no warning channel: what it does not
# understand it passes over silently, so the allowlist is what keeps a picture
# to what was measured. The font is not a package: Dockerfile.linux installs
# InterVariable.ttf from the Inter 4.1 release, pinned by its sha256. MEASURED
# on the pinned base: pytakumi adds 11 MB of site-packages, where the stack it
# replaced added about 115 MB.
pytakumi==0.1.5
