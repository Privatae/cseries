#!/usr/bin/env python3
"""The go-ahead detector, and the verdict no_go_ahead returns.

No cluster, no model. The positives start from three Host replies that end a
turn asking for a one-word go-ahead; the negatives from a reply that reports
background work without asking. Every rule in
go_ahead.RULES has a case below that FAILS if the rule is deleted, every input
of the probe sets is pinned (asks to the rule that must catch
them, finished reports with the property that makes them not an ask), and each
verdict sentence is pinned so the causes cannot collapse into one another.
"""
import json as _json
import sys
from types import SimpleNamespace as _NS

from . import go_ahead
from .report import FAIL, PASS

failures = []


def check(label, cond, extra=""):
    print(("  PASS  " if cond else "  FAIL  ") + label)
    if not cond:
        print(f"        {extra}")
        failures.append(label)


# Labels, one per principle (go_ahead.RULES).
L_OFFER, L_WAIT, L_WORD = go_ahead.L_OFFER, go_ahead.L_WAIT, go_ahead.L_WORD
L_QUESTION, L_LEAD = go_ahead.L_QUESTION, go_ahead.L_LEAD
# Old names the verdict half of this file uses.
A_WORD, O_WISH, H_WAIT, H_QUESTION = L_WORD, L_OFFER, L_WAIT, L_QUESTION

# 1. Three replies that ask for a one-word go-ahead. Each must be caught.
EXAMPLE_ASKS = [
    ("…Anything that fails the floor gets one repair pass before it goes up. "
     "Say go, or swap any concept you don't like.", L_WORD),
    ("…then you can approve the phase in the Workbench — say build and "
     "I'll kick that off.", L_WORD),
    ("…One `build(action='continue')` on the Elm Lantern phase to patch those "
     "two lines, then the four queued themes run. Say go and I'll kick the fix.",
     L_WORD),
]
for text, want in EXAMPLE_ASKS:
    got = go_ahead.asks_for_go_ahead(text)
    check(f"detected ({want}): {text[-50:]!r}", got == want, f"got {got!r}")

# 2. Replies that must NOT be caught.
EXAMPLE_OK = [
    "Elm Lantern is building on Opus 5.5 — task 4, running in the background. "
    "The other four follow as each lands. I'll report here when the first one "
    "finishes.",
    "Done — the file is in your workspace.",
]
for text in EXAMPLE_OK:
    got = go_ahead.asks_for_go_ahead(text)
    check(f"not detected: {text[:50]!r}", got is None, f"got {got!r}")

# 3. One case per principle, so deleting any rule fails here.
PER_RULE = [
    (L_WORD, "Say “go” when you're ready."),
    (L_OFFER, "Done. I can also add a README."),
    (L_WAIT, "Done. Please advise."),
    (L_LEAD, "Created it.\n\nWould you like me to:\n- add a README\n- commit it"),
    (L_QUESTION, "Does that look right?"),
]
covered = {label for label, _ in go_ahead.RULES}
check("every rule has a case here", covered == {l for l, _ in PER_RULE},
      covered ^ {l for l, _ in PER_RULE})
for want, text in PER_RULE:
    got = go_ahead.asks_for_go_ahead(text)
    check(f"rule {want!r}: {text!r}", got == want, f"got {got!r}")

# THE TOOLS WAIT, driven through the in-pod program itself: a fresh Host's
# tools arrive with its identity seed, so the turn waits for them (bounded)
# and the wait is reported.
def _drive_prog(tool_reads, wait_cap=300):
    prog = _sc._NO_GO_AHEAD_PROG
    for k, v in {"__TOKEN__": "'tok'", "__RUN__": "'r1'", "__EID__": "9163", "__ASK__": repr(_sc.NO_GO_AHEAD_ASK),
                 "__TURN__": "240", "__WAIT__": "30", "__TOOLS_WAIT__": str(wait_cap)}.items():
        prog = prog.replace(k, v)
    prog = prog.replace("asyncio.run(main())", "")
    printed, slept, reads = [], [], list(tool_reads)
    ns = {"json": _json, "asyncio": _asyncio, "time": _NS(sleep=lambda s: slept.append(s))}
    exec(prog, ns)
    state = {"turned": False}

    class _PL:
        async def fetchval(self, *a):
            return 0
        async def fetch(self, *a):
            return [{"id": 1, "lane": "host", "tool": "operate", "reason": "ok", "status": "ok", "args": "{}"}]
        async def close(self):
            pass

    async def _pool():
        return _PL()

    async def _tools(eid):
        return dict(reads.pop(0) if len(reads) > 1 else reads[0])

    def _find(tok, name, prefix):
        return ({"code": 200, "total_files": 1, "truncated": False, "root_read": "ok"},
                [name] if state["turned"] else [], [])

    def _stream(*a, **k):
        state["turned"] = True
        return {"text": "Done."}
    ns.update(pool=_pool, _tools_on=_tools, _find=_find, stream=_stream,
              call=lambda *a, **k: {"code": 200, "json": {"ok": True, "content": "x"}},
              print=lambda s: printed.append(s))
    _asyncio.run(ns["main"]())
    return _json.loads(printed[-1]), slept

# mutant pins: a bare-name subject; a word list with no subject; a real
# sentence end before a lower-case word
